} from '../lib/menuTheme';

interface MenuHeaderProps {
  tenant: Tenant;
  location: Location | null;
  onBack?: () => void;
}

// Header estilo delivery app de verdade (McDonald's/FoodyPro): banner
// (foto de capa da loja) e, logo abaixo, nome, status e o cartão de infos de
// entrega.
//
// Celular: desenho vertical original (banner + "sheet" que sobe sobre ele,
// centralizado). Desktop (md+): UM ÚNICO cartão arredondado e com borda que
// encapsula tudo — banner baixo em cima e, embaixo, logo + nome/nota/status à
// esquerda e, à direita, o indicador de unidade e o cartão de entrega
// empilhados. Voltar vira um botão com rótulo sobre o banner.
//
// Todas as cores vêm de lib/menuTheme (par claro/escuro explícito): nenhuma
// superfície fica branca no tema escuro.
export function MenuHeader({ tenant, location, onBack }: MenuHeaderProps) {
  const { t } = useI18n();
  const deliveryAvailable = location?.latitude != null && location?.longitude != null;
  const isOpenNow = useStoreStatus(location).isOpen;

  return (
    // Loja fechada: só banner, logo e o card de entrega ficam em tom atenuado.
    // Selo "Fechado" (vermelho), avaliações e Informações/redes sociais mantêm
    // as cores e continuam clicáveis.
    <div className={MENU_HEADER_SHELL}>
      {/* Banner / capa */}
      <div
        className={`${MENU_BANNER_BASE} h-40 md:h-32 lg:h-36 transition-[filter] ${!isOpenNow ? 'grayscale' : ''}`}
        style={
          tenant.coverImageUrl
            ? undefined
            : {
                background: `linear-gradient(135deg, ${tenant.primaryColor}, ${tenant.secondaryColor})`,
              }
        }
      >
        <BannerViewer tenant={tenant} />

        {/* Voltar (só existe quando há mais de uma unidade): círculo no
            celular; no desktop vira pílula com o rótulo. Sobre a foto, então
            no escuro usa vidro preto translúcido em vez de círculo branco. */}
        {onBack && (
          <button
            onClick={onBack}
            aria-label={t('common.back')}
            className="absolute z-10 top-3.5 left-3.5 md:top-4 md:left-4 h-9 w-9 md:w-auto md:px-3.5 md:gap-1.5 rounded-full bg-white/90 text-[#374151] dark:bg-black/55 dark:text-white backdrop-blur-sm flex items-center justify-center shadow-sm active:scale-90 transition-transform md:hover:bg-white dark:md:hover:bg-black/70"
          >
            <ChevronLeft size={20} strokeWidth={1.5} className="shrink-0" />
            <span className="hidden md:inline text-sm font-medium">{t('common.back')}</span>
          </button>
        )}
      </div>

      {/* Sheet (celular) / corpo do cartão da loja (desktop) */}
      <div className={MENU_HEADER_SHEET}>
        <div className="md:flex md:items-center md:gap-8">
          <div className="flex flex-col items-center text-center md:flex-row md:gap-5 md:text-left md:flex-1 md:min-w-0">
            <div className={`shrink-0 transition-[filter] ${!isOpenNow ? 'grayscale' : ''}`}>
              <LogoViewer tenant={tenant} size="w-16 h-16" />
            </div>

            <div className="flex flex-col items-center md:items-start w-full md:w-auto md:flex-1 md:min-w-0">
              {/* Nome do restaurante SEM truncar: quebra em quantas linhas
                  precisar (text-balance deixa as linhas equilibradas) e
                  `overflowWrap: anywhere` cobre até uma palavra gigante. No
                  celular o botão de QR fica preso no canto direito,
                  centralizado na vertical; o padding lateral simétrico
                  (px-11) reserva o espaço dele nos DOIS lados, então o nome
                  continua centralizado e nunca passa por baixo do botão. No
                  desktop o botão passa a ficar ao lado do nome (em linha). */}
              <div className="relative w-full mt-1.5 md:mt-0 px-11 md:px-0 md:flex md:items-center md:gap-3">
                <h1
                  className={`font-display text-lg md:text-2xl font-bold leading-tight text-center md:text-left text-balance ${MENU_TEXT}`}
                  style={{ overflowWrap: 'anywhere' }}
                >
                  {tenant.name}
                </h1>
                <div className="absolute right-0 top-1/2 -translate-y-1/2 md:static md:translate-y-0 md:shrink-0">
                  <QrScanButton />
                </div>
              </div>
              {/* No celular o nome da unidade fica sob o nome do restaurante;
                  no desktop ele vira o indicador de unidade (ao lado). */}
              {location && <p className={`text-xs mt-0.5 md:hidden ${MENU_TEXT_MUTED}`}>{location.name}</p>}
              <ReviewBadge tenant={tenant} location={location} />
              <OpenStatusRow location={location} />
            </div>
          </div>

          {(location || deliveryAvailable) && (
            <div className={`${deliveryAvailable ? 'mt-3' : ''} md:mt-0 md:w-80 md:shrink-0 flex flex-col gap-2`}>
              {location && <UnitIndicator name={location.name} onChange={onBack} />}

              {deliveryAvailable && location && (
                <div
                  className={`rounded-2xl border px-3.5 py-2.5 flex items-center justify-between gap-3 ${MENU_SURFACE_SUBTLE} ${MENU_BORDER} ${!isOpenNow ? CLOSED_MUTED_CLASS : ''}`}
                >
                  <div className="flex items-center gap-2.5 min-w-0">
                    <span
                      className="w-8 h-8 rounded-full flex items-center justify-center text-white shrink-0"
                      style={{ backgroundColor: tenant.primaryColor }}
                    >
                      <Bike size={15} strokeWidth={1.5} />
                    </span>
                    <div className="min-w-0">
                      <p className={`text-xs font-semibold ${MENU_TEXT_SOFT}`}>{t('menu.acceptsDelivery')}</p>
                      <p className={`text-[11px] ${MENU_TEXT_MUTED}`}>
                        {t('menuHeader.taxa_a_partir_de')} {location.deliveryFee.toFixed(2).replace('.', ',')}
                      </p>
                    </div>
                  </div>
                  {location.minOrderValue > 0 && (
                    <div className={`text-right shrink-0 pl-3 border-l ${MENU_DIVIDER}`}>
                      <p className={`text-[11px] ${MENU_TEXT_MUTED}`}>{t('menu.minimumOrder')}</p>
                      <p className={`text-xs font-bold ${MENU_TEXT_SOFT}`}>
                        R$ {location.minOrderValue.toFixed(2).replace('.', ',')}
                      </p>
                    </div>
                  )}
                </div>
              )}
            </div>
          )}
        </div>

        <RestaurantInfoPanel tenant={tenant} location={location} />
      </div>
    </div>
  );
}

// Indicador de unidade — só no desktop (no celular o nome da unidade já
// aparece sob o nome do restaurante). Com mais de uma unidade (`onChange`
// presente) vira um seletor: tocar leva à tela de escolha de loja, a mesma
// rota do botão Voltar. Com uma unidade só é um rótulo estático.
function UnitIndicator({ name, onChange }: { name: string; onChange?: () => void }) {
  const base = `hidden md:flex items-center gap-2 w-full rounded-xl border px-3 py-2 text-left ${MENU_SURFACE_SUBTLE} ${MENU_BORDER}`;
  const content = (
    <>
      <MapPin size={15} strokeWidth={1.5} className={`shrink-0 ${MENU_TEXT_MUTED}`} />
      <span className={`flex-1 min-w-0 truncate text-sm font-medium ${MENU_TEXT_SOFT}`} title={name}>
        {name}
      </span>
      {onChange && <ChevronDown size={15} strokeWidth={1.5} className={`shrink-0 ${MENU_TEXT_MUTED}`} />}
    </>
  );

  if (!onChange) return <div className={base}>{content}</div>;
  return (
    <button
      type="button"
      onClick={onChange}
      aria-label={name}
      className={`${base} transition-colors md:hover:brightness-95 dark:md:hover:brightness-110`}
    >
      {content}
    </button>
  );
}

      </div>

        <div className="md:flex md:items-center md:gap-8">

            <div className="shrink-0">

            </div>
            <div className="flex flex-col items-center md:items-start w-full md:w-auto md:flex-1 md:min-w-0">
              {/* Nome sem truncar (quebra em várias linhas se preciso, sempre
                  centralizado — padding simétrico px-14 reserva o espaço do
                  timer da mesa, que fica preso no canto esquerdo). */}
              <div className="relative w-full mt-1.5 md:mt-0 px-14 md:px-0 md:flex md:items-center md:gap-3">

                  className={`font-display text-lg md:text-2xl font-bold leading-tight text-center md:text-left text-balance ${MENU_TEXT}`}
                  style={{ overflowWrap: 'anywhere' }}

                      : t('tableMenuHeader.mesa_tablenumber', { tableNumber: tableNumber })
                  : t('tableMenuHeader.consumo_no_local')}

            </div>

import type { ReactNode } from 'react';
import { useI18n } from '../i18n/I18nContext';
import { Link, useLocation } from 'react-router-dom';
import { UtensilsCrossed, ShoppingCart, User, MapPin } from 'lucide-react';
import { useCart } from '../contexts/CartContext';
import { useCustomerAuth } from '../contexts/CustomerAuthContext';
import { useAccentColor } from '../hooks/useAccentColor';
import { useTheme } from '../hooks/useTheme';
import type { Tenant } from '../types';
import { LogoViewer } from './LogoViewer';
import { SearchBar } from './SearchBar';

interface BottomNavProps {
  slug: string;
  qrCodeToken?: string;
  tenantId: string;
  primaryColor?: string;
}

type NavCustomer = ReturnType<typeof useCustomerAuth>['customer'];

// Estado/rotas compartilhados entre a barra inferior (celular) e a barra
// superior (desktop) — uma única fonte de verdade pra hrefs, aba ativa,
// cores e contagem do carrinho.
function useNavModel(slug: string, qrCodeToken: string | undefined, primaryColor: string) {
  const location = useLocation();
  const { totalItems } = useCart();
  const { customer } = useCustomerAuth();
  // No escuro: marca clareada (texto/ícone) e cinza inativo com contraste.
  const accent = useAccentColor(primaryColor);
  const { isDark } = useTheme();
  const inactive = isDark ? '#A3A6A4' : '#9CA3AF';
  const { t } = useI18n();

  const base = qrCodeToken ? `/${slug}/mesa/${qrCodeToken}` : `/${slug}`;
  const menuHref = base;
  const cartHref = `${base}/carrinho`;
  // BUG CORRIGIDO: sem o `?redirect=`, logar/criar conta a partir da
  // mesa sempre devolvia o cliente pro cardápio GERAL (delivery/
  // retirada) em vez de de volta pra própria mesa — aí ele perdia o
  // timer/chamar garçom/pedir conta bem na hora que só queria acessar a
  // conta, e podia acabar tendo que reescanear o QR ou passar pela tela
  // de "você ainda está nessa mesa?" à toa. `menuHref` aqui já é a
  // própria página da mesa quando `qrCodeToken` existe.
  const accountHref = customer
    ? `/${slug}/conta-cliente/perfil`
    : `/${slug}/conta-cliente/entrar?redirect=${encodeURIComponent(menuHref)}`;
  const accountActive = location.pathname.startsWith(`/${slug}/conta-cliente`);

  return {
    t,
    accent,
    inactive,
    customer,
    totalItems,
    menuHref,
    cartHref,
    accountHref,
    accountActive,
    menuActive: location.pathname === menuHref,
    cartActive: location.pathname === cartHref,
  };
}

// Avatar do cliente, igual Instagram — foto de verdade se tiver, senão um
// círculo com a inicial do nome (nunca o ícone genérico de "pessoa").
// Contorno colorido quando a aba está ativa, do mesmo jeito que os outros
// ícones.
function AccountAvatar({ customer, active, accent }: { customer: NavCustomer; active: boolean; accent: string }) {
  return (
    <div
      className="w-6 h-6 rounded-full overflow-hidden shrink-0 flex items-center justify-center"
      style={{
        boxShadow: active ? `0 0 0 1.5px ${accent}` : undefined,
        backgroundColor: customer?.avatarUrl ? undefined : '#D1D5DB',
      }}
    >
      {customer?.avatarUrl ? (
        <img src={customer.avatarUrl} alt="" className="w-full h-full object-cover" />
      ) : customer?.name ? (
        <span className="text-[10px] font-bold text-white">{customer.name[0].toUpperCase()}</span>
      ) : (
        <User size={13} className="text-white" />
      )}
    </div>
  );
}

// Navegação persistente do app do cliente — Cardápio / Carrinho / Conta,
// sempre visível, dentro de UM restaurante (cliente é por restaurante,
// não uma conta cruzando vários — por isso slug/tenantId são sempre
// obrigatórios aqui, nunca "último visitado").
//
// Barra flutuante inferior só no celular (`md:hidden`): em telas grandes ela
// fica redundante e é substituída pela `DesktopNav` no topo da página.
export function BottomNav({ slug, qrCodeToken, primaryColor = '#111827' }: BottomNavProps) {
  const nav = useNavModel(slug, qrCodeToken, primaryColor);

  return (
    <div className="fixed bottom-0 left-0 right-0 max-w-md mx-auto bg-white border-t border-gray-100 flex z-40 md:hidden">
      <NavItem
        to={nav.menuHref}
        icon={UtensilsCrossed}
        label={nav.t('nav.menu')}
        active={nav.menuActive}
        color={nav.accent}
        inactiveColor={nav.inactive}
      />
      <NavItem
        to={nav.cartHref}
        icon={ShoppingCart}
        label={nav.t('nav.cart')}
        active={nav.cartActive}
        color={nav.accent}
        inactiveColor={nav.inactive}
        badgeColor={primaryColor}
        badge={nav.totalItems > 0 ? nav.totalItems : undefined}
      />
      <Link to={nav.accountHref} className="flex-1 flex">
        <div className="flex flex-col items-center gap-1 py-2.5 flex-1 relative">
          <div
            className="relative w-9 h-9 rounded-2xl flex items-center justify-center transition-colors"
            style={{ backgroundColor: nav.accountActive ? `${nav.accent}1a` : 'transparent' }}
          >
            <AccountAvatar customer={nav.customer} active={nav.accountActive} accent={nav.accent} />
          </div>
          <span
            className="text-[10px] font-semibold"
            style={{ color: nav.accountActive ? nav.accent : nav.inactive }}
          >
            {nav.t('nav.account')}
          </span>
        </div>
      </Link>
    </div>
  );
}

interface NavItemProps {
  to: string;
  icon: typeof User;
  label: string;
  active: boolean;
  color: string;
  inactiveColor: string;
  badgeColor?: string;
  badge?: number;
}

function NavItem({ to, icon: Icon, label, active, color, inactiveColor, badgeColor, badge }: NavItemProps) {
  return (
    <Link to={to} className="flex-1 flex">
      <div className="flex flex-col items-center gap-1 py-2.5 flex-1 relative">
        <div
          className="relative w-9 h-9 rounded-2xl flex items-center justify-center transition-colors"
          style={{ backgroundColor: active ? `${color}1a` : 'transparent' }}
        >
          <Icon size={19} color={active ? color : inactiveColor} />
          {Boolean(badge) && (
            <span
              className="absolute -top-1 -right-1.5 text-[9px] font-bold text-white rounded-full w-4 h-4 flex items-center justify-center"
              style={{ backgroundColor: badgeColor ?? color }}
            >
              {badge}
            </span>
          )}
        </div>
        <span className="text-[10px] font-semibold" style={{ color: active ? color : inactiveColor }}>
          {label}
        </span>
      </div>
    </Link>
  );
}

// Equivalente desktop da BottomNav: barra de topo fixa (sticky) no padrão de
// web app — marca à esquerda, busca global no centro, loja/mesa e os mesmos
// três destinos (Cardápio, Carrinho com contador, Conta) à direita. Sem ela,
// o acesso a Carrinho/Conta sumiria quando a barra inferior é ocultada. Só
// aparece a partir de `md`; a altura fixa (h-16) é a referência do `top-16` das
// categorias fixas da MenuPage.
interface DesktopNavProps extends BottomNavProps {
  tenant: Tenant;
  searchValue: string;
  onSearchChange: (value: string) => void;
  contextLabel?: string;
}

export function DesktopNav({
  slug,
  qrCodeToken,
  primaryColor = '#111827',
  tenant,
  searchValue,
  onSearchChange,
  contextLabel,
}: DesktopNavProps) {
  const nav = useNavModel(slug, qrCodeToken, primaryColor);

  return (
    <nav className="hidden md:block sticky top-0 z-40 bg-white border-b border-gray-100">
      <div className="mx-auto w-full max-w-7xl md:px-8 h-16 flex items-center gap-4 lg:gap-6">
        <div className="flex items-center gap-2.5 shrink-0 min-w-0 max-w-[14rem]">
          <LogoViewer tenant={tenant} size="w-9 h-9" />
          <span className="font-display text-sm font-bold text-gray-900 truncate">{tenant.name}</span>
        </div>

        <div className="flex-1 min-w-0 max-w-xl">
          <SearchBar value={searchValue} onChange={onSearchChange} />
        </div>

        <div className="ml-auto flex items-center gap-1 shrink-0">
          {contextLabel && (
            <span className="hidden lg:flex items-center gap-1.5 max-w-[10rem] mr-2 text-xs font-semibold text-gray-700">
              <MapPin size={15} strokeWidth={1.5} className="shrink-0 text-gray-400" />
              <span className="truncate">{contextLabel}</span>
            </span>
          )}
          <DesktopNavItem
            to={nav.menuHref}
            label={nav.t('nav.menu')}
            active={nav.menuActive}
            color={nav.accent}
            inactiveColor={nav.inactive}
            icon={<UtensilsCrossed size={18} color={nav.menuActive ? nav.accent : nav.inactive} />}
          />
          <DesktopNavItem
            to={nav.cartHref}
            label={nav.t('nav.cart')}
            active={nav.cartActive}
            color={nav.accent}
            inactiveColor={nav.inactive}
            icon={
              <span className="relative flex">
                <ShoppingCart size={18} color={nav.cartActive ? nav.accent : nav.inactive} />

function abbreviateState(state: string): string {
  return BR_STATE_ABBREVIATIONS[state.trim().toLowerCase()] ?? state;
}

// Pedido do Felipe (19/09): "Rua X, Bairro" numa linha, "Cidade -
// Estado, CEP" na outra (estado abreviado, sem o prefixo "Cidade:").
// O endereço geocodificado sempre vem como "rua, bairro, cidade,
// estado, CEP" (ver GeocodingService no backend) — quando bate esse
// formato exato, separa bonito; caso venha diferente (endereço
// digitado manualmente antes de existir geocodificação, por exemplo),
// mostra inteiro numa linha só em vez de arriscar cortar errado.
function formatAddressLines(address: string): string[] {
  const parts = address.split(',').map((p) => p.trim());
  if (parts.length === 5) {
    const [street, neighborhood, city, state, zip] = parts;
    return [`${street}, ${neighborhood}`, `${city} - ${abbreviateState(state)}, ${zip}`];
  }
  return [address];
}

// Telefone brasileiro legível: "(48) 99999-9999" (celular), "(48) 3333-4444"
// (fixo), "4003-1234" (sem DDD) e "0800 123 4567". Aceita o número já com
// máscara, com +55/55 na frente ou só dígitos; qualquer coisa que não bata
// com esses formatos volta INTACTA (nunca inventa nem corta dígito). Só
// afeta o que é MOSTRADO — o link `tel:` continua usando o valor cru.
function formatPhoneBR(raw: string): string {
  const digits = raw.replace(/\D/g, '');
  if (digits.length === 11 && /^0[38]00/.test(digits)) {
    return `${digits.slice(0, 4)} ${digits.slice(4, 7)} ${digits.slice(7)}`;
  }
  const national = digits.length >= 12 && digits.startsWith('55') ? digits.slice(2) : digits;
  if (national.startsWith('0')) return raw;
  if (national.length === 11) return `(${national.slice(0, 2)}) ${national.slice(2, 7)}-${national.slice(7)}`;
  if (national.length === 10) return `(${national.slice(0, 2)}) ${national.slice(2, 6)}-${national.slice(6)}`;
  if (national.length === 9) return `${national.slice(0, 5)}-${national.slice(5)}`;
  if (national.length === 8) return `${national.slice(0, 4)}-${national.slice(4)}`;
  return raw;
}

// "1h 20min" / "45min" / "2 dias" — countdown de verdade. Acima de 48h
// vira dias (evita mostrar algo tipo "600h restantes", que ninguém lê
// como tempo de verdade). Atualiza a cada minuto.
function useCountdownLabel(endsAt: string | null): string | null {
  const { t } = useI18n();
  const [label, setLabel] = useState<string | null>(null);

  useEffect(() => {
    if (!endsAt) {
      setLabel(null);
      return;
    }
    function tick() {
      const diffMs = new Date(endsAt!).getTime() - Date.now();
      if (diffMs <= 0) {
        setLabel(t('promoCards.encerrada'));
        return;
      }
      const totalHours = diffMs / 3600000;
      if (totalHours >= 48) {
        const days = Math.floor(totalHours / 24);
        setLabel(`${days} dias`);
        return;
      }
      const hours = Math.floor(diffMs / 3600000);
      const minutes = Math.floor((diffMs % 3600000) / 60000);
      setLabel(hours > 0 ? t('promoCards.hours_h_minutes_min', { hours: hours, minutes: minutes }) : `${minutes}min`);
    }
    tick();
    const interval = setInterval(tick, 60000);
    return () => clearInterval(interval);
  }, [endsAt]);

  return label;
}

  const tableSessionCtx = useTableSessionContext();
  const session = tableSessionCtx?.session ?? null;

  const { tenant, isLoading: isTenantLoading } = useTenant();
  const [categories, setCategories] = useState<Category[]>([]);
  const [products, setProducts] = useState<Product[]>([]);
  const [promotions, setPromotions] = useState<Promotion[]>([]);
  const [showSplash, setShowSplash] = useState(true);
  const [isLoadingMenu, setIsLoadingMenu] = useState(true);
  const [loadError, setLoadError] = useState<string | null>(null);

  const [activeCategoryId, setActiveCategoryId] = useState<string | null>(null);
  // Seções recolhidas dentro da aba "Todos" (só vale nessa aba).
  const [collapsedCategoryIds, setCollapsedCategoryIds] = useState<Set<string>>(new Set());
  const [searchTerm, setSearchTerm] = useState('');
  const [showPromotions, setShowPromotions] = useState(true);

  const { token: customerToken } = useCustomerAuth();

  // Fluxo geral (balcão/entrega): usa a loja escolhida na tela anterior.
  const {
    location: selectedLocation,
    locations,
    isLoading: isLoadingLocations,
  } = useSelectedLocation(!isTableFlow ? tenant?.id : undefined);

  // Fluxo de mesa: a loja é a da própria mesa escaneada, sem escolha
  // nenhuma — só busca os dados dela (aberto/fechado, horário) pra
  // mostrar no header e bloquear pedido se a loja estiver fechada.
  const [tableLocation, setTableLocation] = useState<Location | null>(null);
  useEffect(() => {
    if (!isTableFlow || !tenant || !session?.table?.locationId) return;
    fetchLocationById(tenant.id, session.table.locationId).then(setTableLocation);
  }, [isTableFlow, tenant, session?.table?.locationId]);

  const activeLocation = isTableFlow ? tableLocation : selectedLocation;

  // Passa o token do cliente logado (se houver) — pra promoção com
  // limite por cliente já voltar marcada como "já usada" — e a loja
  // ativa, já que uma promoção pode ser restrita a lojas específicas
  // (ver PromotionsService.findActiveForPublic).
  useEffect(() => {
    if (!tenant?.id) return;
    // Guarda contra race condition: essa busca roda de novo toda vez que
    // `activeLocation` muda (primeiro sem location, depois com) — sem
    // isso, se a busca sem location demorasse mais que a busca com
    // location, ela sobrescrevia o resultado certo com um incompleto.
    let isStale = false;
    fetchActivePromotions(tenant.id, customerToken, activeLocation?.id)
      .then((result) => {
        if (!isStale) setPromotions(result);
      })
      .catch(() => {
        if (!isStale) setPromotions([]);
      });
    return () => {
      isStale = true;
    };
  }, [tenant?.id, customerToken, activeLocation?.id]);

  // Só usados no fluxo de mesa — chamar garçom não existe sem uma sessão.
  const [isCallingWaiter, setIsCallingWaiter] = useState(false);
  const [isWaiterCallPending, setIsWaiterCallPending] = useState(false);
  const [callWaiterError, setCallWaiterError] = useState<string | null>(null);

  useEffect(() => {
    if (!tenant) return;

    async function loadMenu() {
      try {
        setIsLoadingMenu(true);

        // Aplica a cor do tenant como variável CSS global,
        // pra qualquer componente que use var(--tenant-primary) reagir.
        document.documentElement.style.setProperty(
          '--tenant-primary',
          tenant!.primaryColor,
        );

        const [categoriesData, productsData] = await Promise.all([
          fetchCategories(tenant!.id),
          fetchProducts(tenant!.id),
        ]);
        setCategories(categoriesData);
        setProducts(productsData);
      } catch {
        setLoadError(t('menuPage.nao_foi_possivel_carregar'));
      } finally {
        setIsLoadingMenu(false);
      }
    }

    loadMenu();
  }, [tenant]);

  // Sem loja escolhida ainda (fluxo geral, mais de uma loja) — manda pra
  // tela de escolha antes de mostrar qualquer cardápio.
  useEffect(() => {
    if (isTableFlow || isLoadingLocations || !locations) return;
    if (locations.length > 1 && !selectedLocation) {
      navigate(`/${slug}/escolher-loja`, { replace: true });
    }
  }, [isTableFlow, isLoadingLocations, locations, selectedLocation, slug, navigate]);

  // Atualização "em tempo real": refaz a busca de produtos a cada 15s,
  // silenciosamente (sem mostrar o spinner de novo) — assim, se o admin
  // marcar um adicional como indisponível ou mudar um preço enquanto o
  // cliente está navegando, a mudança aparece sozinha em pouco tempo,
  // sem precisar dar F5.
  useEffect(() => {
    if (!tenant?.id) return;
    const interval = setInterval(() => {
      fetchProducts(tenant.id).then(setProducts).catch(() => {
        // Falha pontual de rede — tenta de novo no próximo ciclo, sem
        // incomodar quem está navegando com um erro na tela.
      });
    }, 15000);
    return () => clearInterval(interval);
  }, [tenant?.id]);

  const filteredProducts = useMemo(() => {
    // Só aparecem produtos de categorias ativas (a API pública já manda
    // só as ativas), na ordem das abas quando está em "Todos".
    const order = new Map(categories.map((c, i) => [c.id, i]));
    return products
      .filter((product) => order.has(product.categoryId))
      .sort((a, b) => order.get(a.categoryId)! - order.get(b.categoryId)!)
      .filter((product) => {
      const matchesCategory =
        activeCategoryId === null || product.categoryId === activeCategoryId;
      const matchesSearch = product.name
        .toLowerCase()
        .includes(searchTerm.toLowerCase());
      return matchesCategory && matchesSearch;
    });
  }, [products, categories, activeCategoryId, searchTerm]);

  async function handleCallWaiter() {
    if (!tenant || !session) return;
    setIsCallingWaiter(true);
    setCallWaiterError(null);
    try {
      await callWaiter(tenant.id, session.id, customerToken);
      setIsWaiterCallPending(true);
    } catch (err) {
      const backendMessage =
        err && typeof err === 'object' && 'response' in err
          ? (err as { response?: { data?: { message?: string } } }).response?.data?.message
          : undefined;
      setCallWaiterError(backendMessage ?? t('cartPage.nao_foi_possivel_chamar'));
    } finally {
      setIsCallingWaiter(false);
    }
  }

  // Enquanto o aviso "garçom chamado" está visível, confere a cada 3s se
  // já foi marcado como atendido no painel do admin — some sozinho assim
  // que isso acontecer, em vez de um timer fixo desconectado da realidade.
  // Corta em 10min como rede de segurança.
  useEffect(() => {
    if (!isTableFlow || !isWaiterCallPending || !tenant || !session) return;

    const startedAt = Date.now();
    const interval = setInterval(async () => {
      if (Date.now() - startedAt > 10 * 60 * 1000) {
        setIsWaiterCallPending(false);
        return;
      }
      try {
        const result = await getWaiterCallStatus(tenant.id, session.id);
        if (result.status === 'atendido' || result.status === null) {
          setIsWaiterCallPending(false);
        }
      } catch {
        // Falha de rede pontual não deve travar o polling nem esconder o
        // aviso incorretamente — só tenta de novo no próximo ciclo.
      }
    }, 3000);

    return () => clearInterval(interval);
  }, [isTableFlow, isWaiterCallPending, tenant, session]);

  const productHref = (productId: string) =>
    isTableFlow ? `/${slug}/mesa/${qrCodeToken}/produto/${productId}` : `/${slug}/produto/${productId}`;
  const promotionHref = (promotionId: string) =>
    isTableFlow ? `/${slug}/mesa/${qrCodeToken}/promocao/${promotionId}` : `/${slug}/promocao/${promotionId}`;
  const cartHref = isTableFlow ? `/${slug}/mesa/${qrCodeToken}/carrinho` : `/${slug}/carrinho`;

  // Segurança extra: a mesa escaneada (pelo token) precisa pertencer ao
  // MESMO tenant identificado pela URL (pelo slug). Isso nunca deveria
  // divergir na prática (o link do QR sempre tem os dois corretos), mas
  // se algum dia divergir — link montado errado, cache velho, etc — é
  // melhor travar aqui com um erro claro do que mostrar a marca de um
  // restaurante com a sessão de mesa de outro.
  //
  // BUG REAL CORRIGIDO: essa checagem disparava mesmo enquanto o
  // `TenantProvider` ainda estava buscando um tenant NOVO (troca de
  // slug) e `tenant` no contexto ainda apontava pro tenant ANTERIOR —
  // ou seja, um falso positivo passageiro sempre que o app trocava de
  // restaurante sem dar reload de página inteira. Agora só considera
  // divergência de verdade depois que o tenant termina de carregar.
  if (isTableFlow && !isTenantLoading && tenant && session && session.tenantId !== tenant.id) {
    return (
      <div className={`flex flex-col items-center justify-center h-screen px-6 text-center ${MENU_PAGE_BG}`}>
        <p className={MENU_TEXT_BODY}>
          {t('menuPage.este_qr_code_nao')}
        </p>
      </div>
    );
  }

  if (loadError) {
    return (
      <div className={`flex flex-col items-center justify-center h-screen px-6 text-center ${MENU_PAGE_BG}`}>
        <p className={MENU_TEXT_BODY}>{loadError}</p>
      </div>
    );
  }

  if (showSplash) {
    return <SplashScreen tenant={tenant} onFinish={() => setShowSplash(false)} />;
  }

  const isWaitingForLocationChoice = !isTableFlow && !isLoadingLocations && locations && locations.length > 1 && !selectedLocation;

  if (
    isLoadingMenu ||
    !tenant ||
    (!isTableFlow && isLoadingLocations) ||
    isWaitingForLocationChoice
  ) {
    return (
      <div className={`flex items-center justify-center h-screen ${MENU_PAGE_BG}`}>
        <div className="w-8 h-8 border-2 border-[#E5E7EB] border-t-[#6B7280] dark:border-white/10 dark:border-t-[#C4C7C5] rounded-full animate-spin" />
      </div>
    );
  }

  const isOpenNow = activeLocation?.isOpenNow ?? true;

  return (
    <div className={`min-h-screen ${MENU_PAGE_BG} pb-24 md:pb-28 relative`}>
      <DesktopNav
        slug={slug!}
        qrCodeToken={qrCodeToken}
        tenantId={tenant.id}
        primaryColor={tenant.primaryColor}
        tenant={tenant}
        searchValue={searchTerm}
        onSearchChange={setSearchTerm}
        contextLabel={activeLocation?.name}
      />

      <div className="w-full max-w-7xl mx-auto md:px-8 md:pt-4">
        {isTableFlow ? (
          <TableMenuHeader
            tenant={tenant}
            location={activeLocation}
            tableNumber={session?.table?.number}
            tableKind={session?.table?.kind}
            onCallWaiter={handleCallWaiter}
            onOpenAccount={() => navigate(`/${slug}/mesa/${qrCodeToken}/conta`)}
            isCallingWaiter={isCallingWaiter}
            session={session}
            onExpiryTick={tableSessionCtx?.recheckExpiry}
          />
        ) : (
          <MenuHeader
            tenant={tenant}
            location={activeLocation}
            onBack={
              locations && locations.length > 1
                ? () => navigate(`/${slug}/escolher-loja`)
                : undefined
            }
          />
        )}

        {/* Avisos do garçom: FloatingNotice (portal no body, fixo no topo) em
            vez de um bloco inline com margem negativa — esse era cortado pela
            aba "Informações" e se mexia quando ela abria. */}
        {/* "Garçom chamado": mesmo visual do aviso do QR, mas FIXO — só some
            quando o admin dispensa o chamado no painel (o polling abaixo
            zera isWaiterCallPending). */}
        {isTableFlow && isWaiterCallPending && (
          <FloatingNotice
            persistent
            message={t('menuPage.garcom_chamado_alguem_vai')}
          />
        )}
        {isTableFlow && callWaiterError && (
          <FloatingNotice message={callWaiterError} onDone={() => setCallWaiterError(null)} />
        )}

        {!isOpenNow && (
          <div className="bg-red-50 border-b border-red-100 dark:bg-red-500/10 dark:border-red-500/20 md:mt-4 md:rounded-2xl md:border px-4 py-2.5 text-center">
            <p className="text-xs font-bold text-red-600 dark:text-red-300">{t('store.closedEstablishment')}</p>
            <p className="text-[11px] text-red-500 dark:text-red-300/80 mt-0.5">{t('store.closedHint')}</p>
          </div>
        )}

        {/* Loja fechada: tudo continua navegável (setas, carrossel, busca, abrir
            item/promoção para ler). Só o VISUAL dos cards fica atenuado e os
            botões de adicionar/pedir ficam travados (ver ProductCard e telas de
            detalhe). */}
        <div>
          <div className="pt-1 pb-5">
            {/* Mesmo cabeçalho das categorias da aba "Todos": ícone à
                esquerda + nome + seta de recolher (decisão do Felipe, 30/09). */}
            <button
              onClick={() => setShowPromotions((v) => !v)}
              aria-label={showPromotions ? t('menuPage.ocultar_promocoes') : t('menuPage.mostrar_promocoes')}
              aria-expanded={showPromotions}
              className={SECTION_HEADER_CLASS}
            >
              <PromocoesIcon size={16} />
              <span className="text-sm font-semibold flex-1 text-left">{t('notifications.type.promotion.label')}</span>
              <ChevronDown
                size={16}
                strokeWidth={1.5}
                className={`transition-transform ${MENU_TEXT_MUTED} ${showPromotions ? '' : '-rotate-90'}`}
              />
            </button>

            {showPromotions &&
              (promotions.length > 0 ? (
                <PromoCards
                  promotions={promotions}
                  primaryColor={tenant.primaryColor}
                  muted={!isOpenNow}
                  buildShareUrl={(id) => buildShareUrl('promocao', slug!, id)}
                  onSelect={(id) => navigate(promotionHref(id))}
                />
              ) : (
                <p className={`text-xs px-4 md:px-0 pt-1 pb-1 ${MENU_TEXT_MUTED}`}>{t('menuPage.nenhuma_promocao_no_momento')}</p>
              ))}
          </div>

          <div className="md:hidden">
            <SearchBar value={searchTerm} onChange={setSearchTerm} />
          </div>
          {/* Desktop: categorias fixas logo abaixo da barra superior (h-16). */}
          <div className={`md:sticky md:top-16 md:z-30 ${MENU_PAGE_BG}`}>
            <CategoryChips
              categories={categories}
              activeCategoryId={activeCategoryId}
              onSelect={setActiveCategoryId}
              primaryColor={tenant.primaryColor}
            />
          </div>

          {filteredProducts.length === 0 ? (
            <p className={`text-center text-sm py-12 ${MENU_TEXT_MUTED}`}>
              {t('menuPage.nenhum_produto_encontrado')}
            </p>
          ) : activeCategoryId !== null ? (
            <div className={`${PRODUCT_GRID_CLASS} pt-3 pb-4`}>
              {filteredProducts.map((product) => (
                <ProductCard
                  key={product.id}
                  product={product}
                  primaryColor={tenant.primaryColor}
                  storeClosed={!isOpenNow}
                  shareUrl={buildShareUrl('produto', slug!, product.id)}
                  onClick={() => navigate(productHref(product.id))}
                />
              ))}
            </div>
          ) : (
            // Aba "Todos": itens separados por categoria (só as que o admin
            // ativou), cada seção com ícone + nome e seta pra recolher.
            // Durante uma busca as seções ficam sempre abertas.
            <div className="pt-1 pb-4">
              {categories.map((category) => {
                const items = filteredProducts.filter((p) => p.categoryId === category.id);
                if (items.length === 0) return null;
                const Icon = getCategoryIcon(category.key, category.name);
                const isCollapsed =
                  searchTerm.trim() === '' && collapsedCategoryIds.has(category.id);
                return (
                  <section key={category.id}>
                    <button
                      onClick={() =>
                        setCollapsedCategoryIds((prev) => {
                          const next = new Set(prev);
                          if (next.has(category.id)) next.delete(category.id);
                          else next.add(category.id);
                          return next;
                        })
                      }
                      className={SECTION_HEADER_CLASS}
                      aria-expanded={!isCollapsed}
                    >
                      {Icon && <Icon size={16} />}
                      <span className="text-sm font-semibold flex-1 text-left">{categoryLabel(category, t)}</span>
                      <ChevronDown
                        size={16}
                        strokeWidth={1.5}
                        className={`transition-transform ${MENU_TEXT_MUTED} ${isCollapsed ? '-rotate-90' : ''}`}
                      />
                    </button>
                    {!isCollapsed && (
                      <div className={`${PRODUCT_GRID_CLASS} pt-1.5`}>
                        {items.map((product) => (
                          <ProductCard
                            key={product.id}
                            product={product}
                            primaryColor={tenant.primaryColor}
                            storeClosed={!isOpenNow}
                            shareUrl={buildShareUrl('produto', slug!, product.id)}
                            onClick={() => navigate(productHref(product.id))}
                          />
                        ))}
                      </div>
                    )}
                  </section>
                );
              })}
            </div>
          )}
        </div>
      </div>

      {isOpenNow && (
        <CartBar primaryColor={tenant.primaryColor} onClick={() => navigate(cartHref)} />
      )}

      <BottomNav slug={slug!} qrCodeToken={qrCodeToken} tenantId={tenant.id} primaryColor={tenant.primaryColor} />
    </div>
  );
}
